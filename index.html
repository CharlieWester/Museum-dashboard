# ---- Compute the ratios (same logic as original) ----
museum_data <- Millcompmuseums %>%
  mutate(
    Visitors_per_SqFt        = Visitors / SqFt,
    Visitors_per_1000_MSA    = Visitors / (MSA_Pop / 1000),
    Visitors_per_1000_Local_Tourists = Visitors / (Local_Tourists / 1000),
    Revenue_per_Visitor       = Program_Revenue / Visitors,
    Over_50k_SqFt             = if_else(SqFt > 50000, 1, 0),
    Net_Inventory_Income_Per_Visitor   = Inv_Income / Visitors,
    Rev_per_Visitor        = Rev / Visitors,
    Visitors_per_1000_Pop15 = Visitors / (pop_15 / 1000),
    Visitors_per_1000_Pop30 = Visitors / (pop_30 / 1000),
    Visitors_per_1000_Pop60 = Visitors / (pop_60 / 1000),
    Net_Income_per_rev = (Rev - Expe) / Rev
  )

# ================== SERVER ==================
server <- function(input, output, session) {

  observeEvent(input$select_all, {
    updateCheckboxGroupInput(session, "museum_filter",
                              selected = sort(unique(museum_data$Museum)))
  })

  observeEvent(input$deselect_all, {
    updateCheckboxGroupInput(session, "museum_filter", selected = character(0))
  })

  observeEvent(input$my_peer_group, {
    peer_group <- c("National Brewery Musuem", "National Museum of Industrial History ",
                     "National Quilt Museum", "American Museum of Magic ")
    matched <- intersect(peer_group, museum_data$Museum)
    updateCheckboxGroupInput(session, "museum_filter", selected = matched)
  })

  filtered_data <- reactive({
    req(input$museum_filter, input$pragmatic_filter, input$msa_filter,
        input$sqft_filter, input$interest_filter)
    museum_data %>%
      filter(
        Museum %in% input$museum_filter,
        Pragmatic_Focus %in% as.numeric(input$pragmatic_filter),
        Under_1M_MSA %in% as.numeric(input$msa_filter),
        Over_50k_SqFt %in% as.numeric(input$sqft_filter),
        Interest_Group_Association %in% as.numeric(input$interest_filter)
      )
  })

  output$summaryTable <- renderTable({

    df <- filtered_data()

    tibble(
      Metric = c(
        "Visitors per Sq Ft",
        "Visitors per 1,000 MSA Population",
        "Visitors per 1,000 Population (15 min)",
        "Visitors per 1,000 Population (30 min)",
        "Visitors per 1,000 Population (60 min)",
        "Visitors per 1,000 Local Tourists",
        "Program Revenue per Visitor",
        "Net Inventory Income per Visitor"
      ),

      Mean = c(
        mean(df$Visitors_per_SqFt, na.rm = TRUE),
        mean(df$Visitors_per_1000_MSA, na.rm = TRUE),
        mean(df$Visitors_per_1000_Pop15, na.rm = TRUE),
        mean(df$Visitors_per_1000_Pop30, na.rm = TRUE),
        mean(df$Visitors_per_1000_Pop60, na.rm = TRUE),
        mean(df$Visitors_per_1000_Local_Tourists, na.rm = TRUE),
        mean(df$Revenue_per_Visitor, na.rm = TRUE),
        mean(df$Net_Inventory_Income_Per_Visitor, na.rm = TRUE)
      ),

      Median = c(
        median(df$Visitors_per_SqFt, na.rm = TRUE),
        median(df$Visitors_per_1000_MSA, na.rm = TRUE),
        median(df$Visitors_per_1000_Pop15, na.rm = TRUE),
        median(df$Visitors_per_1000_Pop30, na.rm = TRUE),
        median(df$Visitors_per_1000_Pop60, na.rm = TRUE),
        median(df$Visitors_per_1000_Local_Tourists, na.rm = TRUE),
        median(df$Revenue_per_Visitor, na.rm = TRUE),
        median(df$Net_Inventory_Income_Per_Visitor, na.rm = TRUE)
      ),

      Min = c(
        min(df$Visitors_per_SqFt, na.rm = TRUE),
        min(df$Visitors_per_1000_MSA, na.rm = TRUE),
        min(df$Visitors_per_1000_Pop15, na.rm = TRUE),
        min(df$Visitors_per_1000_Pop30, na.rm = TRUE),
        min(df$Visitors_per_1000_Pop60, na.rm = TRUE),
        min(df$Visitors_per_1000_Local_Tourists, na.rm = TRUE),
        min(df$Revenue_per_Visitor, na.rm = TRUE),
        min(df$Net_Inventory_Income_Per_Visitor, na.rm = TRUE)
      ),

      Max = c(
        max(df$Visitors_per_SqFt, na.rm = TRUE),
        max(df$Visitors_per_1000_MSA, na.rm = TRUE),
        max(df$Visitors_per_1000_Pop15, na.rm = TRUE),
        max(df$Visitors_per_1000_Pop30, na.rm = TRUE),
        max(df$Visitors_per_1000_Pop60, na.rm = TRUE),
        max(df$Visitors_per_1000_Local_Tourists, na.rm = TRUE),
        max(df$Revenue_per_Visitor, na.rm = TRUE),
        max(df$Net_Inventory_Income_Per_Visitor, na.rm = TRUE)
      )

    )

  }, digits = 2)


  output$modelFitTable <- renderTable({
    df <- filtered_data()
    req(nrow(df) >= 3)

    sqft_model  <- tryCatch(lm(Visitors ~ SqFt, data = df), error = function(e) NULL)
    msa_model   <- tryCatch(lm(Visitors ~ MSA_Pop, data = df), error = function(e) NULL)
    local_model <- tryCatch(lm(Visitors ~ Local_Tourists, data = df), error = function(e) NULL)

    pop15_model <- tryCatch(
      lm(Visitors ~ pop_15, data = df),
      error = function(e) NULL
    )

    pop30_model <- tryCatch(
      lm(Visitors ~ pop_30, data = df),
      error = function(e) NULL
    )

    pop60_model <- tryCatch(
      lm(Visitors ~ pop_60, data = df),
      error = function(e) NULL
    )

    r2_sqft  <- if (!is.null(sqft_model))  summary(sqft_model)$r.squared  else NA
    r2_msa   <- if (!is.null(msa_model))   summary(msa_model)$r.squared   else NA
    r2_local <- if (!is.null(local_model)) summary(local_model)$r.squared else NA

    r2_pop15 <- if (!is.null(pop15_model)) summary(pop15_model)$r.squared else NA
    r2_pop30 <- if (!is.null(pop30_model)) summary(pop30_model)$r.squared else NA
    r2_pop60 <- if (!is.null(pop60_model)) summary(pop60_model)$r.squared else NA

    n_sqft  <- if (!is.null(sqft_model))  nobs(sqft_model)  else NA
    n_msa   <- if (!is.null(msa_model))   nobs(msa_model)   else NA
    n_pop15 <- if (!is.null(pop15_model)) nobs(pop15_model) else NA
    n_pop30 <- if (!is.null(pop30_model)) nobs(pop30_model) else NA
    n_pop60 <- if (!is.null(pop60_model)) nobs(pop60_model) else NA
    n_local <- if (!is.null(local_model)) nobs(local_model) else NA

    tibble(
      Predictor = c(
        "Exhibit Size",
        "MSA Population",
        "Population (15 min)",
        "Population (30 min)",
        "Population (60 min)",
        "Local Tourists"
      ),

      R2 = c(
        r2_sqft,
        r2_msa,
        r2_pop15,
        r2_pop30,
        r2_pop60,
        r2_local
      ),

      N = c(
        n_sqft,
        n_msa,
        n_pop15,
        n_pop30,
        n_pop60,
        n_local
      )
    )
  })

  output$estimateTable <- renderTable({
    df <- filtered_data()
    req(nrow(df) > 0)

    set.seed(42)
    n_boot <- 2000

    # ---- Helper: bootstrap a ratio-based estimate ----
    boot_estimate <- function(ratio_vec, input_value, per_thousand = FALSE) {
      ratio_vec <- ratio_vec[!is.na(ratio_vec)]
      if (length(ratio_vec) < 2) return(c(median = NA, lower = NA, upper = NA))
      scale_factor <- if (per_thousand) input_value / 1000 else input_value
      boot_medians <- replicate(n_boot, {
        median(sample(ratio_vec, length(ratio_vec), replace = TRUE)) * scale_factor
      })
      c(median = median(ratio_vec) * scale_factor,
        lower = quantile(boot_medians, 0.025, names = FALSE),
        upper = quantile(boot_medians, 0.975, names = FALSE))
    }

    sqft_est  <- boot_estimate(df$Visitors_per_SqFt, input$est_sqft)
    msa_est   <- boot_estimate(df$Visitors_per_1000_MSA, input$est_msa, per_thousand = TRUE)
    local_est <- boot_estimate(df$Visitors_per_1000_Local_Tourists, input$est_local_tourists, per_thousand = TRUE)

    # Combined estimate: median of the three method-level point estimates,
    # with a pooled bootstrap CI built by resampling within each method and averaging.
    boot_combined <- replicate(n_boot, {
      s <- if (length(na.omit(df$Visitors_per_SqFt)) >= 2)
        median(sample(na.omit(df$Visitors_per_SqFt), replace = TRUE)) * input$est_sqft else NA
      m <- if (length(na.omit(df$Visitors_per_1000_MSA)) >= 2)
        median(sample(na.omit(df$Visitors_per_1000_MSA), replace = TRUE)) * (input$est_msa / 1000) else NA
      l <- if (length(na.omit(df$Visitors_per_1000_Local_Tourists)) >= 2)
        median(sample(na.omit(df$Visitors_per_1000_Local_Tourists), replace = TRUE)) * (input$est_local_tourists / 1000) else NA
      mean(c(s, m, l), na.rm = TRUE)
    })
    combined_median <- median(c(sqft_est["median"], msa_est["median"], local_est["median"]), na.rm = TRUE)
    combined_lower  <- quantile(boot_combined, 0.025, names = FALSE)
    combined_upper  <- quantile(boot_combined, 0.975, names = FALSE)

    tibble(
      Method = c("Based on median visitors/sq ft",
                 "Based on median visitors/1,000 MSA pop.",
                 "Based on median visitors/1,000 local tourists",
                 "Combined estimate (median of the three)"),
      `Estimated Visitors` = c(sqft_est["median"], msa_est["median"], local_est["median"], combined_median),
      `95% CI Lower` = c(sqft_est["lower"], msa_est["lower"], local_est["lower"], combined_lower),
      `95% CI Upper` = c(sqft_est["upper"], msa_est["upper"], local_est["upper"], combined_upper)
    )
  }, digits = 0)

  output$revenueEstimateTable <- renderTable({
    df <- filtered_data()
    req(nrow(df) > 0)

    set.seed(42)
    n_boot <- 2000

    median_revenue_ratio   <- median(df$Revenue_per_Visitor, na.rm = TRUE)
    median_inventory_ratio <- median(df$Net_Inventory_Income_Per_Visitor, na.rm = TRUE)

    boot_combined <- replicate(n_boot, {
      s <- if (length(na.omit(df$Visitors_per_SqFt)) >= 2)
        median(sample(na.omit(df$Visitors_per_SqFt), replace = TRUE)) * input$est_sqft else NA
      m <- if (length(na.omit(df$Visitors_per_1000_MSA)) >= 2)
        median(sample(na.omit(df$Visitors_per_1000_MSA), replace = TRUE)) * (input$est_msa / 1000) else NA
      l <- if (length(na.omit(df$Visitors_per_1000_Local_Tourists)) >= 2)
        median(sample(na.omit(df$Visitors_per_1000_Local_Tourists), replace = TRUE)) * (input$est_local_tourists / 1000) else NA
      mean(c(s, m, l), na.rm = TRUE)
    })
    est_visitors <- median(boot_combined, na.rm = TRUE)
    visitors_lower <- quantile(boot_combined, 0.025, names = FALSE)
    visitors_upper <- quantile(boot_combined, 0.975, names = FALSE)

    est_revenue <- est_visitors * median_revenue_ratio
    est_income  <- est_visitors * median_inventory_ratio
    est_total   <- est_income + est_revenue

    revenue_lower <- visitors_lower * median_revenue_ratio
    revenue_upper <- visitors_upper * median_revenue_ratio
    income_lower  <- visitors_lower * median_inventory_ratio
    income_upper  <- visitors_upper * median_inventory_ratio
    total_lower   <- revenue_lower + income_lower
    total_upper   <- revenue_upper + income_upper

    tibble(
      Metric = c("Estimated Visitors", "Estimated Program Revenue ($)",
                 "Estimated Inventory Income ($)", "Estimated Total Financial Contribution ($)"),
      Estimate = c(est_visitors, est_revenue, est_income, est_total),
      `95% CI Lower` = c(visitors_lower, revenue_lower, income_lower, total_lower),
      `95% CI Upper` = c(visitors_upper, revenue_upper, income_upper, total_upper)
    )
  }, digits = 2)

  output$sqftPlot <- renderPlot({
    ggplot(filtered_data(), aes(x = reorder(Museum, Visitors_per_SqFt),
                                 y = Visitors_per_SqFt)) +
      geom_col(fill = "steelblue") +
      coord_flip() +
      labs(x = NULL, y = "Visitors / sq ft") +
      theme_minimal(base_size = 14) +
      theme(plot.margin = margin(10, 20, 10, 10))
  }, res = 96)

  output$msaPlot <- renderPlot({

    df <- filtered_data()

    variable <- switch(
      input$population_metric,

      "30 Minute Population" = "Visitors_per_1000_Pop30",
      "60 Minute Population" = "Visitors_per_1000_Pop60"
    )

    ggplot(
      df,
      aes(
        x = reorder(Museum, .data[[variable]]),
        y = .data[[variable]]
      )
    ) +
      geom_col(fill = "darkorange") +
      coord_flip() +
      labs(
        x = NULL,
        y = "Visitors per 1,000 population"
      ) +
      theme_minimal(base_size = 14)

  })

  output$localTouristPlot <- renderPlot({
    df <- filtered_data() %>% filter(!is.na(Visitors_per_1000_Local_Tourists))
    ggplot(df, aes(x = reorder(Museum, Visitors_per_1000_Local_Tourists),
                    y = Visitors_per_1000_Local_Tourists)) +
      geom_col(fill = "seagreen") +
      coord_flip() +
      labs(x = NULL, y = "Visitors / 1,000 local tourists") +
      theme_minimal(base_size = 14) +
      theme(plot.margin = margin(10, 20, 10, 10))
  }, res = 96)

  output$expenses <- renderPlot({
    df <- filtered_data() %>% filter(!is.na(Net_Income_per_rev))
    ggplot(df, aes(x = reorder(Museum, Net_Income_per_rev), y = Net_Income_per_rev)) +
      geom_col(fill = "hotpink4") +
      coord_flip() +
      labs(x = NULL, y = "Surplus Margin") +
      theme_minimal(base_size = 14) +
      theme(plot.margin = margin(10, 20, 10, 10))
  }, res = 96)

  output$financialScatter <- renderPlot({

    df <- filtered_data() %>%
      mutate(
        Financial_Contribution =
          (Revenue_per_Visitor + Net_Inventory_Income_Per_Visitor)
      ) %>%
      filter(
        !is.na(Revenue_per_Visitor),
        !is.na(Net_Inventory_Income_Per_Visitor)
      )
    ggplot(
      df,
      aes(
        x = Financial_Contribution,
        y = Net_Income_per_rev,
        color = (Revenue_per_Visitor + Net_Inventory_Income_Per_Visitor) / Rev_per_Visitor
      )
    ) +
      geom_point(size = 4, alpha = 0.8) +
      scale_color_viridis_c(name = "ratio of financial contribution", option = "turbo") +

      geom_text_repel(
        aes(label = Museum),
        seed = 42,
        size = 3.5
      ) +

      geom_vline(
        xintercept = median(df$Financial_Contribution, na.rm = TRUE),
        linetype = "dashed",
        color = "steelblue",
        linewidth = 1
      ) +

      geom_hline(
        yintercept = median(df$Net_Income_per_rev, na.rm = TRUE),
        linetype = "dashed",
        color = "firebrick",
        linewidth = 1
      ) +
      scale_y_continuous(
        labels = scales::percent_format(accuracy = 1)
      ) +
      scale_x_continuous(
        labels = scales::dollar_format()

        title = "Visitor Revenue vs. Operating Surplus",

        x = "Financial Contribution per Visitor",

      theme_minimal(base_size = 14)

  })

  output$financialPlot <- renderPlot({

    df <- filtered_data() %>%
      filter(
        !is.na(Revenue_per_Visitor),
        !is.na(Net_Inventory_Income_Per_Visitor)
      ) %>%
      mutate(
        Total = Revenue_per_Visitor + Net_Inventory_Income_Per_Visitor
      ) %>%
      arrange(Total) %>%
      mutate(Museum = factor(Museum, levels = Museum)) %>%
      pivot_longer(
        c(Revenue_per_Visitor, Net_Inventory_Income_Per_Visitor),
        names_to = "Type",
        values_to = "Value"
      )
    ggplot(df, aes(x = Museum, y = Value, fill = Type)) +
      geom_col() +
      coord_flip() +
      scale_fill_manual(values = c("purple4", "gold1"),
                         labels = c("Inventory Income", "Program Revenue")) +
      labs(x = NULL, y = "Contribution per Visitor ($)", fill = NULL) +
      theme_minimal(base_size = 14)

  }, res = 96)

  output$TotalRevPlot <- renderPlot({
    df <- filtered_data() %>% filter(
      !is.na(Revenue_per_Visitor),
      !is.na(Net_Inventory_Income_Per_Visitor),
      !is.na(Rev_per_Visitor)
    ) %>%
      mutate(totalRatio = (Revenue_per_Visitor + Net_Inventory_Income_Per_Visitor) / Rev_per_Visitor)

    ggplot(df, aes(x = reorder(Museum, totalRatio),
                   y = totalRatio)) +
      geom_col(fill = "royalblue3") +
      coord_flip() +
      labs(x = NULL, y = "Financial Contriubtion / total rev / visitor") +
      theme_minimal(base_size = 14) +
      theme(plot.margin = margin(10, 20, 10, 10))
  }, res = 96)

  output$scatterPlot <- renderPlot({

    df <- filtered_data()

    predictor <- switch(
      input$scatter_predictor,
      "Exhibit Size" = "SqFt",
      "MSA Population" = "MSA_Pop",
      "Population (15 min)" = "pop_15",
      "Population (30 min)" = "pop_30",
      "Population (60 min)" = "pop_60",
      "Local Tourists" = "Local_Tourists"
    )

    x_label <- switch(
      input$scatter_predictor,
      "Exhibit Size" = "Exhibit Space (sq ft)",
      "MSA Population" = "MSA Population",
      "Population (15 min)" = "Population within 15 Minutes",
      "Population (30 min)" = "Population within 30 Minutes",
      "Population (60 min)" = "Population within 60 Minutes",
      "Local Tourists" = "Local Tourists"
    )

    # Remove missing values
    df <- df %>%
      filter(
        !is.na(.data[[predictor]]),
        !is.na(Visitors)
      )

    # Fit regression model
    model <- lm(
      log10(Visitors) ~ log10(df[[predictor]]),
      data = df
    )

    r2 <- round(summary(model)$r.squared, 3)

    ggplot(
      df,
      aes(
        x = log10(.data[[predictor]]),
        y = log10(Visitors)
      )
    ) +

      geom_point(
        size = 4,
        alpha = 0.8,
        color = "steelblue"
      ) +

      geom_text_repel(
        aes(label = Museum),
        size = 3.5,
        box.padding = 0.4,
        point.padding = 0.2,
        max.overlaps = Inf,
        seed = 42
      ) +

      geom_smooth(
        method = "lm",
        formula = y ~ x,
        se = TRUE,
        color = "firebrick",
        linewidth = 1
      ) +
      scale_x_continuous(
        labels = function(x) scales::comma(10^x)
      ) +
      scale_y_continuous(
        labels = function(y) scales::comma(10^y)
      ) +
      annotate(
        "text",
        x = Inf,
        y = Inf,
        label = paste0("R\u00b2 = ", r2),
        hjust = 1.1,
        vjust = 1.5,
        size = 5,
        fontface = "bold"
      ) +

      labs(
        title = "Attendance vs Selected Predictor",
        x = x_label,
        y = "Annual Visitors"
      ) +

      theme_minimal(base_size = 14)

  })
